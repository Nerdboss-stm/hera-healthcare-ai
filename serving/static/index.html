Quick Start
Use the tabs above to interact with each pipeline, or call the API directly.
# Summarize a clinical note
curl -X POST http://localhost:8000/api/summarize \
  -H "Content-Type: application/json" \
  -d '{"note": "Patient is a 65yo male with chest pain..."}'

# Predict patient risk from vitals
curl -X POST http://localhost:8000/api/predict \
  -H "Content-Type: application/json" \
  -d '{"heart_rate":95,"respiratory_rate":22,"body_temperature":38.1,
       "oxygen_saturation":93,"systolic_bp":145,"diastolic_bp":92,"age":67}'

# Upload a .txt clinical note file
curl -X POST http://localhost:8000/api/summarize/upload \
  -F "file=@clinical_note.txt"
